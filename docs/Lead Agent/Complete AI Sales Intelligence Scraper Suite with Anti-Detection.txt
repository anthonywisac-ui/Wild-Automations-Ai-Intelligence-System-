async def human_like_click(page, selector str)
    Click with random offset and Bézier mouse movement.
    box = await page.locator(selector).bounding_box()
    if not box
        return
    x = box[x] + box[width]  random.uniform(0.2, 0.8)
    y = box[y] + box[height]  random.uniform(0.2, 0.8)
    # Simulate curved mouse path
    await page.mouse.move(x, y, steps=random.randint(5, 15))
    await asyncio.sleep(random.uniform(0.1, 0.3))
    await page.mouse.down()
    await asyncio.sleep(random.uniform(0.05, 0.1))
    await page.mouse.up()

async def human_like_scroll(page, delta int = 500)
    Scroll with variable speed and random pauses.
    steps = random.randint(10, 30)
    step_delta = delta  steps
    for _ in range(steps)
        await page.mouse.wheel(delta_y=step_delta  random.uniform(0.5, 1.5))
        await asyncio.sleep(random.uniform(0.02, 0.08))

        self.headless = config.get(headless, False)  # False = more human‑like
        self.proxies = get_proxy_list()
        self.proxy = random.choice(self.proxies) if self.proxies else None
        self.session_name = config.get(session_name, stealth_session)
        self.use_stealth = config.get(use_stealth, True)

    async def _get_browser_context(self)
        Create a browser context with stealth patches and proxy.
        from playwright.async_api import async_playwright
        p = await async_playwright().start()
        # Choose browser (Chromium is best for stealth)
        browser = await p.chromium.launch(
            headless=self.headless,
            proxy={server self.proxy} if self.proxy else None,
            args=[
                --disable-blink-features=AutomationControlled,
                --disable-features=IsolateOrigins,site-per-process,
                --disable-site-isolation-trials,
                --no-sandbox,
            ]
        )
        context = await browser.new_context(
            user_agent=get_random_user_agent(),
            viewport={width random.choice([1280, 1366, 1440, 1920]), height random.choice([720, 768, 900, 1080])},
            locale=en-US,
            timezone_id=AmericaNew_York,
            permissions=[geolocation],
            extra_http_headers={
                Accept-Language en-US,en;q=0.9,
                Accept-Encoding gzip, deflate, br,
                Referer httpswww.google.com,
            }
        )
        # Apply stealth patches if available
        if self.use_stealth
            try
                from playwright_stealth import stealth_async
                page = await context.new_page()
                await stealth_async(page)
                await page.close()
            except ImportError
                pass  # stealth not installed, proceed with basic
        return browser, context

    async def fetch_leads(self, tenant_id UUID, limit int = 10, kwargs) - List[LeadRecord]
        raise NotImplementedError

    def _make_lead(self, tenant_id UUID, name str, source_url str, source str,

            email=email,
            phone=phone,
            company=company,
            title=title,
            source=source,
            source_url=source_url,
            enrichment_data=enrichment or {},
        )

class SocialAnalyzer
    Uses OpenRouter (LLM) to analyze postscommentsreviews.
    def __init__(self, config Dict[str, Any])
        self.api_key = config.get(openrouter_key) or os.getenv(OPENROUTER_API_KEY)
        self.model = config.get(model, openaigpt-3.5-turbo)
        self.timeout = config.get(timeout, 45)

    async def analyze(self, text str, context Dict[str, Any] = None) - Dict[str, Any]
        if not self.api_key
            raise ValueError(OpenRouter API key missing)
        import openai
        client = openai.OpenAI(
            base_url=httpsopenrouter.aiapiv1,
            api_key=self.api_key,
        )
        prompt = f
        Analyze the following social media postreview and extract
        - Buying signals (e.g., looking for, need, upgrade, considering)
        - Sentiment (positiveneutralnegative)
        - Intent (interested, not_interested, undecided)
        - A one-sentence summary
        - Whether this person is likely a potential lead (actionable truefalse)

        Content {text[1500]}

        Return ONLY JSON with keys buying_signals (list), sentiment, intent, summary, actionable.
        
        try
            response = await asyncio.to_thread(
                client.chat.completions.create,
                model=self.model,
                messages=[{role user, content prompt}],
                response_format={type json_object},
                temperature=0.3,
                max_tokens=300,
                timeout=self.timeout,
            )
            return json.loads(response.choices[0].message.content)
        except Exception as e
            print(fOpenRouter error {e})
            return {}

    }

            all_leads.extend(leads)
        else

    # Deduplicate by source_url

    unique_leads = []

        if lead.source_url not in seen
            seen.add(lead.source_url)
            unique_leads.append(lead)

    # Analyze if requested
    if analyze
        analyzer = SocialAnalyzer({})
        for lead in unique_leads
            text = 
            if lead.source == linkedin
                text = lead.enrichment_data.get(linkedin_profile, {}).get(about, )
            elif lead.source == instagram
                text = lead.enrichment_data.get(instagram_profile, {}).get(biography, )
            elif lead.source == facebook
                text = lead.enrichment_data.get(post, {}).get(message, )
            elif lead.source == google_maps
                text = lead.enrichment_data.get(place, {}).get(formatted_address, )
            if text

                    analysis = await analyzer.analyze(text)
                    lead.enrichment_data[analysis] = analysis
                    if analysis.get(actionable)
                        lead.score += 30
                    if analysis.get(sentiment) == positive
                        lead.score += 20
                    if analysis.get(intent) == interested
                        lead.score += 25
                    lead.score = min(100, lead.score)
                    if lead.score = 70

                except Exception as e
                    print(fAnalysis failed for {lead.source_url} {e})
    return unique_leads

class LinkedInScraper:

    async def _fetch_agentic(self, tenant_id: UUID, limit: int, keyword: str) -> List[LeadRecord]:
