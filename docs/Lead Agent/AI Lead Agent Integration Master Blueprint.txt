from datetime import datetime
from typing import Optional, Dict, Any, List
from uuid import UUID, uuid4
from pydantic import BaseModel, EmailStr, Field

class LeadRecord(BaseModel):
    """Canonical lead record - this is the ONLY schema the system accepts."""
    
    # Identifiers
    lead_id: UUID = Field(default_factory=uuid4)
    tenant_id: UUID
    
    # Core Identity
    name: str = Field(..., min_length=1, max_length=255)
    email: Optional[EmailStr] = None
    phone: Optional[str] = None
    
    # Firmographic
    company: Optional[str] = Field(None, max_length=255)
    title: Optional[str] = Field(None, max_length=255)
    industry: Optional[str] = None
    company_size: Optional[int] = None
    
    # Source Metadata
    source: str  # e.g., "linkedin", "google_maps", "web_crawl"
    source_url: Optional[str] = None  # UNIQUE - prevents duplicates
    crawled_at: datetime = Field(default_factory=datetime.utcnow)
    
    # Enrichment Data (flexible JSONB in PostgreSQL)
    enrichment_data: Dict[str, Any] = Field(default_factory=dict)
    """
    Expected structure:
    {
        "linkedin_profile": {
            "work_history": [...],
            "skills": [...],
            "education": [...]
        },
        "social_posts": [...],
        "scraped_site_metadata": {
            "technologies": [...],
            "keywords": [...],
            "description": "..."
        },
        "contact_info": {
            "emails": [...],
            "phones": [...]
        }
    }
    """
    
    # Scoring
    score: float = 0.0
    status: str = "new"  # new|contacted|qualified|lost
    
    # Audit
    created_at: datetime = Field(default_factory=datetime.utcnow)
    updated_at: datetime = Field(default_factory=datetime.utcnow)

# In lead_scoring_engine.py - add your scoring logic

async def calculate_icp_score(lead: LeadRecord) -> float:
    """
    Calculate ICP (Ideal Customer Profile) score.
    Returns: Float between 0.0 and 100.0
    """
    score = 50.0  # Base score
    
    # Example scoring logic
    if lead.title:
        title_scores = {
            "cto": 30, "vp": 25, "director": 20,
            "head": 20, "manager": 15, "engineer": 10,
            "founder": 35, "ceo": 30
        }
        for keyword, points in title_scores.items():
            if keyword in lead.title.lower():
                score += points
                break
    
    # Company size bonus
    if lead.company_size:
        if 50 <= lead.company_size <= 500:
            score += 15
        elif 10 <= lead.company_size < 50:
            score += 5
    
    # Industry bonus
    target_industries = ["saas", "fintech", "ai", "healthtech", "edtech"]
    if lead.industry and lead.industry.lower() in target_industries:
        score += 20
    
    return min(100.0, score)

def _page_new(self, cursor: Optional[str] = None) -> Dict[str, Any]:
    """Get next page of results using cursor-based pagination."""
    params = self._base_params.copy()
    if cursor:
        params["cursor"] = cursor
    response = self._make_request(params)
    return {
        "data": response["data"],
        "next_cursor": response.get("next_cursor")
    }

async def scrape_and_publish(tenant_id: UUID, config: Dict[str, Any]):
    leads = await fetch_leads(config, tenant_id)
    
    for lead in leads:
        # 1. Publish to Kafka
        await event_dispatcher.dispatch(
            LeadCreatedEvent(lead=lead, tenant_id=tenant_id)

            index="leads",
            id=str(lead.lead_id),
            body=lead.model_dump()
        )

Q: How do I handle pagination?

async def fetch_leads(config, **kwargs) -> List[LeadRecord]:
    results = []
    cursor = None
    
    while True:
        page_data = await _page_new(cursor)
        results.extend(page_data["data"])
        
        cursor = page_data.get("next_cursor")
        if not cursor:
            break
        
        # Respect rate limit
        await asyncio.sleep(0.1)
    
    return results

# Run with test data
poetry run python -m crm.plugins.scrapers.your_scraper --test --limit 5

# Mock external dependencies
poetry run pytest tests/unit/test_scrapers/test_your_scraper.py --mock-external
Q: What if my scraper fails?
python
try:
    leads = await fetch_leads(config)
except ScrapingError as e:
    # Log error
    logger.error("Scraper failed: %s", e, exc_info=True)
    
    # Move to Dead Letter Queue
    await dlq_service.send({
        "scraper": "your_scraper",
        "config": config,
        "error": str(e)
    })
    
    # Notify team
    await alert_service.send("Scraper failed", e)
    
    return []
