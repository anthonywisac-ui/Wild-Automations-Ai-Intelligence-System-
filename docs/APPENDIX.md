3.2 Docker Compose Override
yaml
# docker-compose.override.yml
version: '3.8'

services:
  api:
    environment:
      - DEBUG=true
      - LOG_LEVEL=DEBUG
      - ENVIRONMENT=development
    volumes:
      - ./src:/app/src
      - ./tests:/app/tests
    command: uvicorn main:app --host 0.0.0.0 --port 8000 --reload

  worker:
    environment:
      - DEBUG=true
      - LOG_LEVEL=DEBUG
      - ENVIRONMENT=development
    volumes:
      - ./src:/app/src
    command: celery -A tasks worker --loglevel=debug

  postgres:
    ports:
      - "5432:5432"
    environment:
      - POSTGRES_DB=wildautomations
      - POSTGRES_USER=wildautomations
      - POSTGRES_PASSWORD=password

  redis:
    ports:
      - "6379:6379"

  pgadmin:
    image: dpage/pgadmin4
    environment:
      - PGADMIN_DEFAULT_EMAIL=admin@wildautomations.com
      - PGADMIN_DEFAULT_PASSWORD=admin
    ports:
      - "5050:80"
    depends_on:
      - postgres

4.2 Troubleshooting
yaml
Common Issues and Solutions:

  Database Connection:
    Issue: "Could not connect to database"
    Solution:
      - Check DB_HOST and DB_PORT
      - Verify credentials
      - Check network connectivity
      - Ensure database is running

  Redis Connection:
    Issue: "Could not connect to Redis"
    Solution:
      - Check REDIS_HOST and REDIS_PORT
      - Verify Redis is running
      - Check authentication
      - Verify network connectivity

  API Gateway Errors:
    Issue: "Rate limit exceeded"
    Solution:
      - Check rate limiting configuration
      - Implement exponential backoff
      - Verify user authentication
      - Check API quotas

  Scraper Issues:
    Issue: "Scraping failed"
    Solution:
      - Check source configuration
      - Verify URL accessibility
      - Check robots.txt
      - Verify authentication
      - Check IP restrictions

  AI Service Issues:
    Issue: "AI model unavailable"
    Solution:
      - Check API key validity
      - Verify model availability
      - Check rate limits
      - Implement fallback strategy
